```C
int removeElement(int* nums, int numsSize, int val) { //双指针 
	int left = 0, right = numsSize;
	while (left < right){
		if (nums[left] == val) {
			nums[left] = nums[right - 1];
			right--;
		}
		else {
			left++;
		}
	}
	return left;
}
```

3. 逆向双指针
```java
class Solution {
    public void merge(int[] nums1, int m, int[] nums2, int n) {
        int p1 = m - 1, p2 = n - 1;
        int tail = m + n - 1;
        int cur;
        while (p1 >= 0 || p2 >= 0) {
            if (p1 == -1) {
                cur = nums2[p2--];
            }
            else if (p2 == -1) {
                cur = nums1[p1--];
            }
            else if (nums1[p1] > nums2[p2]) {
                cur = nums1[p1--];
            }
            else {
                cur = nums2[p2--];
            }
            nums1[tail--] = cur;
        }
    }
}
```

```C
int searchInsert(int* nums, int numsSize, int target) {
       int left = 0, right = numsSize - 1, ans = numsSize;
    while (left <= right) {
        int mid = ((right - left) >> 1) + left;
        if (target <= nums[mid]) {
            ans = mid;
            right = mid - 1;
        } else {
            left = mid + 1;
        }
    }
    return ans;
}
```

## 牛顿迭代公式（x的平方根）
- 我的
```java
int mySqrt(int x) {
	if (x >= 0 && x < 4) {
		if (x == 0) {
			return 0;
		}
		else {
			return 1;
		}
	}
	unsigned int i = 2;
	while (i <= (x / 2)) {
		if (i * i > x) {
			return --i;
			break;
		}
		else {
			i++;
		}
	}
	return --i;
}
```
- 牛顿迭代公式
Newton迭代法的基本思想是利用函数在某一点的切线来近似函数图形，并通过不断迭代逼近函数的零点。
```java
int mySqrt(int x) { 
	long x0=x; //注意，这里必须是long类型，防止下面x0*x0溢出 
	while( x0 * x0 > x ) { 
		x0 = (x0 + x / x0) / 2;//牛顿迭代公式 
	} 
	return (int)x0; 
}
```

1. **选择初始点**：选择一个初始近似值 $𝑥_0$。

2. **计算导数**：在 $𝑥_k$ 点计算函数 𝑓(𝑥) 的导数 $f'(𝑥)$。
    
3. **构造切线**：在点 $(𝑥_k,𝑓(𝑥_k))$ 处,函数的切线方程为:$𝑦=𝑓'(𝑥_k)(𝑥−𝑥_k)+𝑓(𝑥_k)$

4. **求切线与x轴的交点**：令 𝑦=0,解上述切线方程得到 $𝑥_{k+1}:𝑥_{k+1}=𝑥_k−𝑓(𝑥_k)/𝑓'(𝑥_k)$
    
5. **迭代**：重复步骤2到4，直到满足一定的精度要求，即 $|𝑥_{k+1}−𝑥_k|$ 小于预设的阈值。
- Newton迭代法的收敛速度通常比其他数值方法快，但它有一个前提条件：函数 𝑓(𝑥) 在零点附近必须可导，且导数不为零。

![[Pasted image 20240720171853.png|300]]

如果用到这题就是，x是已知的，x0是我们假设的也就是未知数，一开始先给x0一个近似值（符号范围即可，答案给的是x）。
这题的条件是x0 * x0 > x，所以我们建立的模型就是$y=x0^2-x$ ，对该函数进行计算，就得到$x0 = x0 - f(x0)/f'(x0)$，（注意等号左边的和等号右边的x0不同）化简得$x0 = (x0 + x / x0) / 2$

- 二分查找
```java
int mySqrt(int x) { 
	if (x == 0) return 0; 
	int l = 1; 
	long r = x; 
	while (l < r) { 
		long mid = (l + r + 1) >> 1; //这里相当于除以2
		if((long long)mid * mid <= x) l = mid; 
		//如果mid的平方小于x，就将左边界移到mid，因为最小的都比mid大
		else {
			r = mid - 1;//否则将右边界移到mid-1，因为平方根不大于mid
		}
	} 
	return l; 
}
```

## 矩阵快速幂
### 快速幂
- 比如说 $2^{10}$，假如使用 10 个 2 相乘，需要 10 次操作
- 使用快速幂
	- 将 10 转化为二进制，$(1010)_{2}$，有 4 位
	- 我们只需要计算出最大位数的即可，$2^8$
		- 因为计算最大位数的过程就能得到其他所有的幂，相当于只需要四次操作
		- 也就是 log 10 次
- 最后时间复杂度：n-->logn


### 矩阵快速幂
- 将数字换成矩阵


![[Pasted image 20240725131626.png]]

-核心：斐波那契数列

- 注意：不能用递归，会超出时间限制
-我的：（这个方法适用于n比较小的情况，如果比较大则不适用）
```C
int climbStairs(int n) {
	int count = 0;
	int  pre_plus= 1;
	int pre = 2;
	if (n <= 2) {
		return n;
	}
	for (int i = 3; i <= n; i++) {
		count = pre_plus + pre;
		pre_plus = pre;
		pre = count;
	}
	return count;
}
```

- 矩阵快速幂:（$O(log\space n)$）

- 以斐波那契数列为例
	- 可以先看看这段解释：[[Lec 06 特征值，特征向量#用在斐波那契数列中|斐波那契数列]]，[快速幂解释](https://oi-wiki.org/math/binary-exponentiation/)
	- 复习之后我们可以得出一个这样的式子：
$$\left[\begin{matrix} f(n+1)\\f(n)\end{matrix}\right]=\left[\begin{matrix} 1&1\\1&0\end{matrix}\right]^n\left[\begin{matrix} f(1)\\f(0)\end{matrix}\right],A=\left[\begin{matrix} 1&1\\1&0\end{matrix}\right]$$
我们只需要计算A的n次幂就就可以得到f(n)的值，我们可以定义矩阵乘法，然后用快速幂算法来加速$A^n$的求取。
- 如何想到矩阵快速幂？
	1. 如果一个问题可以转化为一个矩阵的n次方形式，就可以使用快速幂来加速
	2. 如果一个递归式形如$f(n)=\sum^m_{i=1}a_if(n-i)$，即齐次线性递推式，我们就可以将数列的递推关系转化为矩阵的递推关系，构造一个$u_{k+1} = Au_{k}$,再转换为$u_k=A^ku_0$的形式。(u是列向量)
-  一般情况下，形如上面递归式可以构造出m x m的矩阵：$$\left[\begin{matrix} a_1&a_2&a_3&\cdots&a_m\\
   1&0&0&\cdots&0\\
   0&1&0&\cdots&0\\
   0&0&1&\cdots&0\\
   \vdots&\vdots&\vdots&\ddots&\vdots\\
   0&0&0&0&1
   \end{matrix}\right]$$
   - 对非齐次线性递归的处理
	   - 可以先将非齐次线性递推转化为齐次线性递推：例如这样一个递推：$f(x)=(2x-6)+f(x-1)+f(x-2)+f(x-3)$
先进行这样的变换：$f(x)+xc=[f(x-1)+(x-1)c]+[f(x-2)+(x-2)c]+[f(x-3)+(x-3)c],$$令g(x)=f(x)+xc$，那么我们就得到了一个齐次线性递推：$g(x)=g(x-1)+g(x-2)+g(x-3)$

- 思考题：你能把 f(x)=2f(x−1)+3f(x−2)+4c 化成齐次线性递推吗？
$f(x)+c = 2(f(x-1)+c)+3(f(x-2)+c)$
齐次方程：$g(x)=f(x)+c$
$g(x)=2g(x-1)+3g(x-2)$

时间复杂度：$O(log\space n)$
空间复杂度：$O(1)$
```C
struct Matrix {
    long long mat[2][2];//矩阵结构体
};

struct Matrix multiply(struct Matrix a, struct Matrix b) {
    struct Matrix c;
    for (int i = 0; i < 2; i++) {
        for (int j = 0; j < 2; j++) {
            c.mat[i][j] = a.mat[i][0] * b.mat[0][j] + a.mat[i][1] * b.mat[1][j];//c的ij位=a的i行*b的j列
        }
    }
    return c;
}

struct Matrix matrixPow(struct Matrix a, int n) {//快速幂算法
    struct Matrix ret;
    ret.mat[0][0] = ret.mat[1][1] = 1;
    ret.mat[0][1] = ret.mat[1][0] = 0;//单位矩阵
    while (n > 0) {
        if ((n & 1) == 1) {//判断二进制最低位是否为1，如果是1就就要分解独立出来，说该位有数，如果是0说明没有就不需要
            ret = multiply(ret, a);
        }
        n >>= 1;//除以2的意思，右移一位，最低位处理完换2的平方为，依次类推
        a = multiply(a, a);
    }
    return ret;
}

int climbStairs(int n) {
    struct Matrix ret;
    ret.mat[1][1] = 0;
    ret.mat[0][0] = ret.mat[0][1] = ret.mat[1][0] = 1;//A矩阵
    struct Matrix res = matrixPow(ret, n);//将A矩阵相乘
    return res.mat[0][0];
}
```

```java
public class Solution {  
    public int climbStairs(int n) {  
        int[][] q = {{1, 1}, {1, 0}};  
        int[][] res = pow(q, n);  
        return res[0][0];  
    }  
  
    public int[][] pow(int[][] a, int n) {  
        int[][] ret = {{1, 0}, {0, 1}};  
        while (n > 0) {  
            if ((n & 1) == 1) {  
                ret = multiply(ret, a);  
            }  
            n >>= 1;  
            a = multiply(a, a);  
        }  
        return ret;  
    }  
  
    public int[][] multiply(int[][] a, int[][] b) {  
        int[][] c = new int[2][2];  
        for (int i = 0; i < 2; i++) {  
            for (int j = 0; j < 2; j++) {  
                c[i][j] = a[i][0] * b[0][j] + a[i][1] * b[1][j];  
            }  
        }  
        return c;  
    }  
}
```

- 解释为什么返回`res.mat[0][0]`：
	- A的第一行表示到达当前台阶和上一台阶的方法数之和
	- 第二行表示到达当前台阶的方法数
$$A^n的表示的含义是\left[\begin{matrix} f(n)&f(n-1)\\f(n-1)&f(n-2)\end{matrix}\right]$$

- 方法三：通项公式
知道A后，我们可以知道特征方程|A-λI|=0=>可以求出λ，接下来就是用线代中的那个方法：通解是$f(n)=c_1​x^n​+c_2x^n$
将f(1)=1,f(2)=1带入，解出$c_1,c_2$得到递推公式：$f(n)=\frac{1}{\sqrt{5}}[(\frac{1+\sqrt5}{2})^n-(\frac{1-\sqrt5}{2})^n]$


- 复杂度分析：代码中使用的 pow 函数的时空复杂度与 CPU 支持的指令集相关，这里不深入分析

```C
int climbStairs(int n) 
{
```
