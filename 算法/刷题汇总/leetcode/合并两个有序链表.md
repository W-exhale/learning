struct ListNode* create_node(int val) {
    struct ListNode* new_node = (struct ListNode*)malloc(sizeof(struct ListNode));

    if (new_node == NULL) {
        printf("Failed to mallocate!");
        return NULL;
    }
    new_node->val = val;
    new_node->next = NULL;
    
    return new_node;
}

void insert(struct ListNode** node, int val){
    struct ListNode* new_node = create_node(val);
    struct ListNode* current = *node;
    while (current->next != NULL) {
        current = current->next;
    }
    current->next = new_node;
}
struct ListNode* mergeTwoLists(struct ListNode* list1, struct ListNode* list2){

    struct ListNode* new_list = NULL;

    if (list1 == NULL && list2 == NULL) {
        return NULL;
    }
    if (list1 == NULL && list2 != NULL) {
        return list2;
    }
    if(list1 != NULL && list2 == NULL){
        return list1;
    }

    if (list1->val <= list2->val) {
        new_list = create_node(list1->val);
        list1 = list1->next;
    }
    else {
        new_list = create_node(list2->val);
        list2 = list2->next;
    }

    struct ListNode* current = new_list;

    while (list1 != NULL || list2 != NULL) {
        if (list1 == NULL) {
            while (current->next != NULL) {
                current = current->next;
            }
            current->next = list2;
        }
        if (list2 == NULL) {
            while (current->next != NULL) {
                current = current->next;
            }
            current->next = list1;
        }
        if (list1 != NULL && list2 != NULL) {
            if (list1->val < list2->val) {
                insert(new_list, list1->val);
                list1 = list1->next;
            }
            else if (list1->val == list2->val) {
                insert(&new_list, list1->val);
                insert(&new_list, list1->val);
                list1 = list1->next;
                list2 = list2->next;
            }
            else {
                insert(&new_list, list2->val);
                list2 = list2->next;
            }
        }
        else {
            break;
        }
    }
    return new_list;
}
